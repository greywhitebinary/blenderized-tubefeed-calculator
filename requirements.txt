# Runtime dependencies only - what app/streamlit_app.py and src/ actually

# >>> WORKING ON YOUR OWN MACHINE? Install requirements-dev.txt, not this
# >>> file:   pip install -r requirements-dev.txt
# This file alone is enough for Streamlit Cloud (a fresh clone has no
# Parquet cache, so src/data_loader.py takes its CSV fallback and pyarrow
# is never needed). But a LOCAL checkout usually DOES have a built cache
# under data/processed/, and data_loader.py calls pd.read_parquet() as
# soon as those files exist - with no pyarrow installed that raises an
# ImportError on startup. Installing the dev file avoids it entirely.

streamlit==1.58.0
openpyxl>=3.1

# Label-photo reading (src/label_extract.py). Pinned to a major version:

# change under us, and it is the one that spends money.
anthropic>=0.120,<1.0

pillow>=10.0
